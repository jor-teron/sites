/* =========================================================
   WHERE THE LAND ENDS
   Below sea level you stand on a narrow shore and the rest of the
   scene is water. The forest is always the same forest — we only
   hide the trees that would be standing in the sea.
   Above sea level the whole scene is land (the valley is drawn
   underneath), so every tree shows.
   ========================================================= */
function layoutTrees() {
  const groundW = Math.max(1, groundEl.clientWidth);

  /* --water-span is written by ground.js: 0 on land, 0.45 at the Dead Sea.
     Trees get 90 % of the shore, so none of them stand in the water. */
  const land = 1 - waterSpan(state.place) * 0.9;

  /* --h/--w are percentages of the ground box, which is taller than it
     used to be; shrink them together when the scene is narrow */
  const narrow = Math.max(0.6, Math.min(1, groundW / 640));

  trees.forEach((t) => {
    const x = t.slot * land;
    const offshore = waterSpan(state.place) > 0 && t.slot * land > land - 0.03;
    t.el.style.display = offshore ? "none" : "";
    if (offshore) return;

    t.el.style.left = (x * 100).toFixed(1) + "%";
    t.el.style.setProperty("--h", (t.height * 62 * t.depth * narrow).toFixed(1) + "%");
    t.el.style.setProperty("--w", (t.height * t.widthRatio * 62 * t.depth * narrow).toFixed(1) + "%");
    t.el.style.opacity = (0.72 + t.depth * 0.28).toFixed(2);
    t.el.style.zIndex = String(Math.round(t.depth * 10));
  });
}
